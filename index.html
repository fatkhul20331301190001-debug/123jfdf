Membaca data dari Google Sheets dan memastikan nominal dibaca sebagai angka async function fetchTransactionsFromSheet() { if (!API_URL) return; try { const response = await fetch(API_URL); const data = await response.json(); if (Array.isArray(data)) { state.transactions = data.map(t => ({ id: Number(t.id) || Date.now(), group: String(t.group || 'mingguan').toLowerCase(), date: String(t.date || ''), type: String(t.type || 'masuk').toLowerCase(), category: String(t.category || ''), // Mengonversi string/undefined menjadi angka murni amount: Number(t.amount) || 0, note: String(t.note || '') })); saveLocalTransactions(); renderData(); } } catch (err) { console.warn("Gagal memuat dari Google Sheets, menggunakan data lokal:", err); renderData(); } } // 2. Menyimpan transaksi baru dari form input nominal async function handleAddTransaction(event) { event.preventDefault(); const btn = document.getElementById('btnSubmitForm'); if (btn) { btn.disabled = true; btn.innerHTML = ' Menyimpan...'; } // Mengambil nilai input dan mengonversi nominal ke Number const rawAmount = document.getElementById('inputAmount').value; const numericAmount = Number(rawAmount) || 0; const newTx = { id: Date.now(), group: document.getElementById('inputGroup').value, date: document.getElementById('inputDate').value, type: document.getElementById('inputType').value, category: document.getElementById('inputCategory').value, amount: numericAmount, // Nominal tersimpan sebagai angka note: document.getElementById('inputNote').value }; // Tambahkan ke state lokal dan render UI state.transactions.unshift(newTx); saveLocalTransactions(); renderData(); resetForm(); // Kirim data ke Google Sheets melalui API if (API_URL) { try { await fetch(API_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8', }, body: JSON.stringify({ action: 'add', data: newTx }) }); } catch (err) { console.error("Gagal sinkronisasi online ke Google Sheets:", err); } } if (btn) { btn.disabled = false; btn.innerHTML = ' Simpan Transaksi'; } }
