// Key unik untuk menyimpan data transaksi di memori browser (localStorage) const STORAGE_KEY = 'infaq_kas_data'; // Data galeri default jika belum ada gambar yang diunggah const defaultGallery = [ { id: 1, title: 'Kegiatan Perbaikan Kelas', url: 'https://placehold.co/400x300/166534/ffffff?text=Perbaikan+Kelas' }, { id: 2, title: 'Pembelian Karpet Musala', url: 'https://placehold.co/400x300/15803d/ffffff?text=Karpet+Musala' } ]; // State utama aplikasi let state = { accessMode: 'public', activeTab: 'all', transactions: [], gallery: defaultGallery }; // 1. MEMUAT DATA SAAT APLIKASI DIBUKA function loadStoredData() { try { const savedTx = localStorage.getItem(STORAGE_KEY); if (savedTx) { state.transactions = JSON.parse(savedTx); } const savedGal = localStorage.getItem('infaq_gallery_data'); if (savedGal) { state.gallery = JSON.parse(savedGal); } } catch (e) { console.warn("Gagal membaca memori lokal:", e); } } // 2. MENYIMPAN DATA KE MEMORI BROWSER (LOCALSTORAGE) function saveToLocalStorage() { try { localStorage.setItem(STORAGE_KEY, JSON.stringify(state.transactions)); } catch (e) { alert("Gagal menyimpan ke memori browser. Kapasitas penyimpanan penuh."); } } // 3. FUNGSI UTAMA: SIMPAN TRANSAKSI BARU function saveTransactionDirectly() { const categoryInput = document.getElementById('inputCategory'); const amountInput = document.getElementById('inputAmount'); const groupInput = document.getElementById('inputGroup'); const dateInput = document.getElementById('inputDate'); const typeInput = document.getElementById('inputType'); const noteInput = document.getElementById('inputNote'); // Validasi sederhana if (!categoryInput.value.trim() || !amountInput.value) { alert('Mohon isi Kategori dan Nominal Transaksi dengan benar.'); return; } // Mengonversi input nominal menjadi angka murni const numericAmount = Number(amountInput.value) || 0; // Membuat objek transaksi baru const newTx = { id: Date.now(), group: groupInput.value, date: dateInput.value, type: typeInput.value, category: categoryInput.value.trim(), amount: numericAmount, note: noteInput.value.trim() }; // Tambahkan ke urutan paling atas state.transactions.unshift(newTx); // Simpan ke localStorage & perbarui tampilan di layar saveToLocalStorage(); renderData(); resetFormFields(); alert('Transaksi berhasil disimpan di browser!'); } // 4. MENGOSONGKAN FORM INPUT SETELAH SIMPAN function resetFormFields() { document.getElementById('inputCategory').value = ''; document.getElementById('inputAmount').value = ''; document.getElementById('inputNote').value = ''; const dateInput = document.getElementById('inputDate'); if (dateInput) dateInput.valueAsDate = new Date(); } // 5. MENGHAPUS TRANSAKSI function deleteTransaction(id) { if (confirm('Apakah Anda yakin ingin menghapus transaksi ini?')) { state.transactions = state.transactions.filter(t => t.id !== id); saveToLocalStorage(); renderData(); } } // 6. FORMAT MATA UANG RUPIAH function formatRp(num) { return 'Rp ' + Number(num || 0).toLocaleString('id-ID'); } // 7. FORMAT TANGGAL (YYYY-MM-DD -> DD/MM/YYYY) function formatDate(dateStr) { if (!dateStr) return '-'; const parts = String(dateStr).slice(0, 10).split('-'); if (parts.length !== 3) return dateStr; return `${parts[2]}/${parts[1]}/${parts[0]}`; } // 8. MENYARING TRANSAKSI SESUAI FILTER & PENCARIAN function getFilteredTransactions() { const filterType = document.getElementById('filterType') ? document.getElementById('filterType').value : 'all'; const keyword = document.getElementById('searchKeyword') ? document.getElementById('searchKeyword').value.toLowerCase().trim() : ''; return state.transactions.filter(item => { const matchTab = state.activeTab === 'all' || item.group === state.activeTab; const matchType = filterType === 'all' || item.type === filterType; const matchSearch = String(item.category || '').toLowerCase().includes(keyword) || String(item.note || '').toLowerCase().includes(keyword); return matchTab && matchType && matchSearch; }); } // 9. MERENDER KARTU SALDO, TABEL, DAN RINCIAN KAS function renderData() { const filtered = getFilteredTransactions(); const tableBody = document.getElementById('transactionTableBody'); // Render Tabel if (tableBody) { tableBody.innerHTML = ''; if (filtered.length === 0) { tableBody.innerHTML = `Belum ada data transaksi tersimpan.`; } else { filtered.forEach((tx, idx) => { const row = document.createElement('tr'); row.className = 'hover:bg-slate-50/80 transition'; row.innerHTML = ` ${idx + 1} ${tx.group} ${formatDate(tx.date)} ${tx.category} ${tx.note || '-'} ${tx.type === 'masuk' ? formatRp(tx.amount) : '-'} ${tx.type === 'keluar' ? formatRp(tx.amount) : '-'} ${state.accessMode === 'admin' ? ` ` : '-'} `; tableBody.appendChild(row); }); } } // Hitung Total Pemasukan & Pengeluaran const activeSet = state.transactions.filter(t => state.activeTab === 'all' || t.group === state.activeTab); const totalMasuk = activeSet.filter(t => t.type === 'masuk').reduce((acc, curr) => acc + (Number(curr.amount) || 0), 0); const totalKeluar = activeSet.filter(t => t.type === 'keluar').reduce((acc, curr) => acc + (Number(curr.amount) || 0), 0); const sisaSaldo = totalMasuk - totalKeluar; if (document.getElementById('cardTotalMasuk')) document.getElementById('cardTotalMasuk').textContent = formatRp(totalMasuk); if (document.getElementById('cardTotalKeluar')) document.getElementById('cardTotalKeluar').textContent = formatRp(totalKeluar); if (document.getElementById('cardSisaSaldo')) document.getElementById('cardSisaSaldo').textContent = formatRp(sisaSaldo); } // 10. MENGUBAH MODE (PUBLIK / PENGURUS) function setAccessMode(mode) { state.accessMode = mode; const adminForm = document.getElementById('adminFormSection'); const adminBtns = document.querySelectorAll('.admin-only-btn'); if (mode === 'admin') { if (adminForm) adminForm.classList.remove('hidden'); adminBtns.forEach(btn => btn.classList.remove('hidden')); } else { if (adminForm) adminForm.classList.add('hidden'); adminBtns.forEach(btn => btn.classList.add('hidden')); } renderData(); } // INISIALISASI SAAT HALAMAN SELESAI DIMUAT document.addEventListener('DOMContentLoaded', () => { loadStoredData(); const dateInput = document.getElementById('inputDate'); if (dateInput) dateInput.valueAsDate = new Date(); setAccessMode('public'); renderData(); });
